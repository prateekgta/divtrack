RedisSubscriberConfig.java

@Bean
public ReactiveRedisMessageListenerContainer
    priceListenerContainer(...) {

  container
    .receive(ChannelTopic.of("prices:live"))
    .map(msg -> msg.getMessage())
    .flatMap(json -> {
      String ticker = objectMapper
        .readTree(json)
        .path("ticker").asText();

      return fanOutHandler
        .fanOut(ticker, json); // ← THE BRIDGE
    })
    .retry() // reconnect if Redis drops
    .subscribe();

  return container;
}

PriceFanOutHandler.java

public Mono<Void> fanOut(
    String ticker, String jsonMsg) {

  var sessions =
    registry.getSessionsFor(ticker);

  if (sessions.isEmpty())
    return Mono.empty();

  return Flux.fromIterable(sessions)
    .flatMap(session ->
      session.send(
        Mono.just(
          session.textMessage(jsonMsg)
        )
      )
      .onErrorResume(e -> {
        registry.deregister(session);
        return Mono.empty();
      })
    ).then();
}

SessionRegistry.java

// ticker → set of live WS sessions
private final Map<String,
  CopyOnWriteArraySet<WebSocketSession>>
  tickerSessions = new ConcurrentHashMap<>();

public void register(
    String ticker, WebSocketSession session) {
  tickerSessions
    .computeIfAbsent(
      ticker.toUpperCase(),
      k -> new CopyOnWriteArraySet<>()
    )
    .add(session);
}

public Set<WebSocketSession>
    getSessionsFor(String ticker) {
  return tickerSessions.getOrDefault(
    ticker.toUpperCase(),
    Collections.emptySet()
  );
}
